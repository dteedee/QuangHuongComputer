# W4-2 — job CI cho bộ test tích hợp (Testcontainers + PostgreSQL thật).
# W4-5 dán khối này vào .github/workflows/ci.yml (W4-2 KHÔNG sửa file workflow).
#
# Ghi chú vận hành:
#  - Job chạy RIÊNG, không song song với job unit test: mỗi lần chạy dựng 1 container Postgres
#    và khởi động toàn bộ host ApiGateway (~100s, ~1.5GB RAM).
#  - Không cần service container nào: Testcontainers tự dựng Postgres 16 và tự dọn.
#  - Ryuk bị tắt có chủ đích; container được IAsyncLifetime dispose.
#  - TESTCONTAINERS cần Docker socket — runner ubuntu-latest của GitHub đã có sẵn.

  integration-tests:
    name: Test tích hợp (API + PostgreSQL thật)
    runs-on: ubuntu-latest
    needs: [backend-tests]
    timeout-minutes: 30
    env:
      DOTNET_CLI_TELEMETRY_OPTOUT: 1
      TESTCONTAINERS_RYUK_DISABLED: "true"
    steps:
      - uses: actions/checkout@v4

      - name: Cài .NET 8
        uses: actions/setup-dotnet@v4
        with:
          dotnet-version: 8.0.x

      - name: Kéo sẵn image Postgres 16
        run: docker pull postgres:16

      - name: Chạy bộ test tích hợp
        run: >-
          dotnet test backend/Tests/IntegrationTests/IntegrationTests.csproj
          -m:1
          --logger "trx;LogFileName=integration-tests.trx"
          --results-directory artifacts/test-results

      - name: Lưu kết quả test
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: integration-test-results
          path: artifacts/test-results
